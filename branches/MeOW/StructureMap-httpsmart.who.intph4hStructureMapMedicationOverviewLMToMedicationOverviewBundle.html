/// url = 'MedicationOverviewLMToMedicationOverviewBundle'
/// name = 'http://smart.who.int/ph4h/StructureMap/MedicationOverviewLMToMedicationOverviewBundle'
/// status = 'draft'

uses "http://smart.who.int/ph4h/StructureDefinition/MedicationOverviewLM" alias MedicationOverviewLM as source
uses "http://smart.who.int/ph4h/StructureDefinition/PatientLM" alias PatientLM as source
uses "http://smart.who.int/ph4h/StructureDefinition/MedicationTreatmentLineLM" alias MedicationTreatmentLineLM as source
uses "http://smart.who.int/ph4h/StructureDefinition/PractitionerLM" alias PractitionerLM as source
uses "http://smart.who.int/ph4h/StructureDefinition/MedicationOverview" alias MedicationOverviewBundle as target
uses "http://smart.who.int/ph4h/StructureDefinition/MedicationOverviewComposition" alias MedicationOverviewComposition as target
uses "http://smart.who.int/ph4h/StructureDefinition/MedicationTreatmentLine" alias MedicationTreatmentLine as target
uses "http://hl7.org/fhir/StructureDefinition/Patient" alias Patient as target
uses "http://hl7.org/fhir/StructureDefinition/Practitioner" alias Practitioner as target // ---------------------------------------------------------------------------

// Top-level group: assemble MedicationOverview Bundle (document)
// ---------------------------------------------------------------------------
group MedicationOverviewLMToMedicationOverviewBundle(source src : MedicationOverviewLM, target tgt : MedicationOverviewBundle) {
  src -> tgt.type = 'document' "bundleType";
  src -> tgt.timestamp = (now()) "bundleTimestamp";
  src ->  uuid() as bid,  uuid() as cid,  uuid() as pid then {
    src ->  tgt.identifier as id,  id.system = 'urn:ietf:rfc:4122',  id.value = bid "bundleIdentifier"; // Bundle.identifier (UUID identifier per RFC 4122)
    src ->  tgt.entry as ce,  ce.fullUrl = append('urn:uuid:', cid),  ce.resource = create('Composition') as comp then {
      src then MedicationOverviewLMToComposition(src, comp, pid) "compositionContent";
    } "compositionEntry"; // Composition entry
    src.patient as pref ->  tgt.entry as pe,  pe.fullUrl = append('urn:uuid:', pid),  pe.resource = create('Patient') as pat then {
      pref then PatientLMToPatient(pref, pat) "patientContent";
    } "patientEntry"; // Patient entry
    src.medicationTreatmentLine as lref -> uuid() as tid then {
      lref ->  tgt.entry as te,  te.fullUrl = append('urn:uuid:', tid),  te.resource = create('MedicationStatement') as ms then {
        lref then MedicationTreatmentLineLMToMedicationStatement(lref, ms, pid) "lineContent";
      } "lineEntry";
    } "treatmentLines"; // Treatment line entries — UUID generated per iteration (variable count)
  } "setEntries"; // Generate fixed UUIDs once; reuse them across entries and Reference targets.
}

// ---------------------------------------------------------------------------
// MedicationOverviewLM -> MedicationOverviewComposition
// Receives the Patient entry's UUID (pid) so subject can reference it.
// ---------------------------------------------------------------------------
group MedicationOverviewLMToComposition(source src : MedicationOverviewLM, target tgt : MedicationOverviewComposition, source pid) {
  src -> tgt.status = 'final' "compStatus";
  src ->  create('Coding') as cc,  cc.system = 'http://loinc.org',  cc.code = '56445-0',  cc.display = 'Medication summary Document',  create('CodeableConcept') as ct,  ct.coding = cc,  tgt.type = ct "compType"; // Composition.type — LOINC 56445-0 "Medication summary Document"
  src ->  create('Reference') as subj,  subj.reference = append('urn:uuid:', pid),  tgt.subject = subj "compSubject"; // Composition.subject -> Patient entry by uuid
  src -> tgt.date = (now()) "compDate";
  src -> tgt.title = 'Medication Overview' "compTitle";
  src -> tgt.section as sect then {
    src -> sect.title = 'Medication Treatment Lines' "sectTitle";
    src ->  create('Coding') as scc,  scc.system = 'http://loinc.org',  scc.code = '10160-0',  scc.display = 'History of Medication use Narrative',  create('CodeableConcept') as sc,  sc.coding = scc,  sect.code = sc "sectCode";
    src.medicationTreatmentLine as lref ->  create('Reference') as ref,  ref.display = 'Medication Treatment Line',  sect.entry = ref "sectEntry"; // Display-only Reference per line — cross-linking to bundle entry would
  } "sectionTreatmentLines"; // Section: MedRecordTreatmentLine
  src.verification as v -> tgt.attester as att then {
    v -> att.mode = 'professional' "attMode";
    v.verificationTime as vt -> att.time = vt "attTime";
    v.practicioner as pr ->  create('Reference') as party,  party.display = (pr.name.text),  att.party = party "attParty";
  } "attester"; // Attester from verification BackboneElement
  src.comment as cm -> create('Annotation') as nt then {
    cm.commentText as ctxt -> nt.text = ctxt "noteText";
    cm.date as d -> nt.time = d "noteTime";
    cm.author as a ->  create('Reference') as ar,  ar.display = (a.name.text),  nt.authorReference = ar "noteAuthor";
    cm -> tgt.note = nt "noteAssign";
  } "noteFromComment"; // Composition.note from comment BackboneElement
}

// ---------------------------------------------------------------------------
// PatientLM -> FHIR Patient
// ---------------------------------------------------------------------------
group PatientLMToPatient(source src : PatientLM, target tgt : Patient) {
  src.identifier as id -> tgt.identifier = id "patIdentifier";
  src.name as n -> tgt.name = n "patName";
  src.gender as g -> tgt.gender = (g.coding.code) "patGender";
  src.dateOfBirth as d -> tgt.birthDate = d "patBirthDate"; // dateOfBirth is dateTime in LM; birthDate is date — silent truncation expected.
}

// ---------------------------------------------------------------------------
// MedicationTreatmentLineLM -> MedicationTreatmentLine (profile on MedicationStatement, R5)
// Receives the Patient entry's UUID (pid) so MS.subject can reference it.
// ---------------------------------------------------------------------------
group MedicationTreatmentLineLMToMedicationStatement(source src : MedicationTreatmentLineLM, target tgt : MedicationTreatmentLine, source pid) {
  src.identifier as id -> tgt.identifier = id;
  src ->  create('Reference') as subj,  subj.reference = append('urn:uuid:', pid),  tgt.subject = subj "subject"; // subject -> Patient entry by uuid (R5 MS.subject is 1..1)
  src.status as st -> tgt.status = st;
  src.medicationReference as mr ->  create('CodeableReference') as m,  m.reference = mr,  tgt.medication = m "medicationRef"; // medication[x] choice -> R5 CodeableReference
  src.medicationCodeableConcept as mc ->  create('CodeableReference') as m,  m.concept = mc,  tgt.medication = m "medicationCC";
  src.effectivePeriod as ep -> tgt.effectivePeriod = ep;
  src.recordingMetadata as rm then {
    rm.recordedTime as rt -> tgt.dateAsserted = rt "dateAsserted";
    rm.recorder as rc -> tgt.informationSource = rc "informationSource";
  };
  src.indication as ind ->  create('CodeableReference') as r,  r.concept = ind,  tgt.reason = r "reasonIndication"; // reason = indication + indicationText + intendedUse
  src.indicationText as it ->  create('CodeableConcept') as cc,  cc.text = it,  create('CodeableReference') as r,  r.concept = cc,  tgt.reason = r "reasonIndicationText";
  src.intendedUse as iu ->  create('CodeableReference') as r,  r.concept = iu,  tgt.reason = r "reasonIntendedUse";
  src.usageInstructions as ui -> tgt.dosage = ui "dosage";
  src.preparationInstructions as pi ->  create('Dosage') as dos,  dos.text = pi,  tgt.dosage = dos "prepInstructions";
  src.comment as cm -> create('Annotation') as nt then {
    cm.commentText as ct -> nt.text = ct "lineNoteText";
    cm.date as d -> nt.time = d "lineNoteTime";
    cm.author as a ->  create('Reference') as ar,  ar.display = (a.name.text),  nt.authorReference = ar "lineNoteAuthor";
    cm -> tgt.note = nt "lineNoteAssign";
  } "lineNote"; // note = comment
  src.treatmentStatus as ts -> tgt.adherence as adh then {
    ts -> adh.code = ts "adherenceCode";
  } "adherence"; // adherence — BackboneElement; navigate.
  src.category as cat -> tgt.category = cat;
  src.derivedFrom as df then {
    df.request as rq -> tgt.derivedFrom = rq "derivedFromRequest";
    df.statement as stm -> tgt.derivedFrom = stm "derivedFromStatement";
    df.dispensation as dsp -> tgt.derivedFrom = dsp "derivedFromDispensation";
    df.administration as adm -> tgt.derivedFrom = adm "derivedFromAdministration";
    df.other as oth -> tgt.derivedFrom = oth "derivedFromOther";
  };
  src.version as v ->  create('Extension') as ext,  ext.url = 'http://hl7.org/fhir/StructureDefinition/artifact-version',  ext.value = v,  tgt.extension = ext "extVersion"; // Extension: artifact-version
  src.verificationInformation as vi -> create('Extension') as ext then {
    vi -> ext.url = 'http://smart.who.int/ph4h/StructureDefinition/VerificationInformation' "viUrl";
    vi.verifier as vr ->  create('Extension') as e1,  e1.url = 'verifier',  e1.value = vr,  ext.extension = e1 "viVerifier";
    vi.verificationTime as vt ->  create('Extension') as e2,  e2.url = 'verificationTime',  e2.value = vt,  ext.extension = e2 "viTime";
    vi -> tgt.extension = ext "viAssign";
  } "extVerification"; // Extension: VerificationInformation
  src.substitution as sub -> create('Extension') as ext then {
    sub -> ext.url = 'http://smart.who.int/ph4h/StructureDefinition/Substitution' "subUrl";
    sub.substitutionType as t ->  create('Extension') as e1,  e1.url = 'substitutionType',  e1.value = t,  ext.extension = e1 "subType";
    sub.substitutionTypeReason as r ->  create('Extension') as e2,  e2.url = 'substitutionTypeReason',  e2.value = r,  ext.extension = e2 "subReason";
    sub.substitutionTypeReasonText as rt ->  create('Extension') as e3,  e3.url = 'substitutionTypeReasonText',  e3.value = rt,  ext.extension = e3 "subReasonText";
    sub -> tgt.extension = ext "subAssign";
  } "extSubstitution"; // Extension: Substitution
}

// ---------------------------------------------------------------------------
// PractitionerLM -> FHIR Practitioner
// ---------------------------------------------------------------------------
group PractitionerLMToPractitioner(source src : PractitionerLM, target tgt : Practitioner) {
  src.identifier as id -> tgt.identifier = id "practIdentifier";
  src.name as n -> tgt.name = n "practName";
}
